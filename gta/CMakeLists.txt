cmake_minimum_required(VERSION 3.20)
project(EldenLosSantosProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(probe_test tests/probe_test.cpp)
target_include_directories(probe_test PRIVATE src)
if(MSVC)
  target_compile_options(probe_test PRIVATE /W4 /WX)
else()
  target_compile_options(probe_test PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
enable_testing()
add_test(NAME probe_state_and_native_abi COMMAND probe_test)
add_executable(combat_test tests/combat_test.cpp)
target_include_directories(combat_test PRIVATE src)
add_test(NAME combat_scenarios COMMAND combat_test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME dds_texture_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/texture_test.py")
add_test(NAME owner_launch_safety COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/launch_test.py")
add_test(NAME native_call_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/native_contract_test.py")

if(WIN32)
  if(NOT MINGW)
    message(FATAL_ERROR "The verified Windows build currently requires MinGW-w64/dlltool")
  endif()
  find_program(DLLTOOL NAMES x86_64-w64-mingw32-dlltool dlltool REQUIRED)
  set(RUNTIME_IMPORT "${CMAKE_CURRENT_BINARY_DIR}/scripthook-import.a")
  add_custom_command(OUTPUT "${RUNTIME_IMPORT}"
    COMMAND "${DLLTOOL}" -d "${CMAKE_CURRENT_SOURCE_DIR}/tools/runtime.def" -l "${RUNTIME_IMPORT}"
    DEPENDS tools/runtime.def VERBATIM)
  add_custom_target(runtime_import DEPENDS "${RUNTIME_IMPORT}")
  add_library(EldenLosSantosProbe SHARED src/plugin.cpp)
  add_library(EldenLosSantos SHARED src/encounter_plugin.cpp)
  add_dependencies(EldenLosSantos runtime_import)
  target_link_libraries(EldenLosSantos PRIVATE "${RUNTIME_IMPORT}")
  target_include_directories(EldenLosSantos PRIVATE src)
  set_target_properties(EldenLosSantos PROPERTIES PREFIX "" SUFFIX ".asi")
  target_compile_options(EldenLosSantos PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_link_options(EldenLosSantos PRIVATE -static -static-libgcc -static-libstdc++)
  add_dependencies(EldenLosSantosProbe runtime_import)
  target_link_libraries(EldenLosSantosProbe PRIVATE "${RUNTIME_IMPORT}")
  target_include_directories(EldenLosSantosProbe PRIVATE src)
  set_target_properties(EldenLosSantosProbe PROPERTIES PREFIX "" SUFFIX ".asi")
  if(MINGW)
    target_compile_options(EldenLosSantosProbe PRIVATE -Wall -Wextra -Werror -Wpedantic)
    target_link_options(EldenLosSantosProbe PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(EldenLosSantosProbe PRIVATE /W4)
  endif()
endif()
